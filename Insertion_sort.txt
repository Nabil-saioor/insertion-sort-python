#The insertion sort
n = input("Enter the number of values: ")
n = int(n)
values = []

for i in range(1,n+1):
    value = input(" Enter the value number "+str(i)+":  ")
    value = int(value)
    values.append(value)

print(" Your sequence befor sorting is: "+str(values)+".")

for j in range(1,n-1):
	i = 0
	while values[j] > values[i]:
		i += 1
	m = values[j]
	for k in range(0,j-i-1):
		values[j-k] = values[j-k-1]
	values[i] = m
		
print(" Your sequence aftr sorting is: "+str(values)+".")
